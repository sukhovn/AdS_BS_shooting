\section{Boson Star equations}
We take following scalar field theory
\begin{equation}
S = \int\left[\frac{R + 2\Lambda}{2\kappa} - |\partial_\mu \phi|^2 - V(|\phi|^2) \right]\sqrt{-g} d^4 x.
\end{equation}
The equations of motions then become
\begin{subequations}
\begin{align}
	&R_{\mu\nu} - \frac{1}{2}R g_{\mu\nu} - \Lambda g_{\mu\nu} = \kappa T_{\mu\nu},\\
	&T_{\mu\nu} = \partial_\mu \phi^* \partial_\nu \phi + \partial_\nu \phi^* \partial_\mu \phi - g_{\mu\nu}(|\partial_\alpha \phi|^2 + V(|\phi|^2)),\\
	&\square \phi - \frac{dV}{d |\phi|^2} \phi = 0.
\end{align}
\end{subequations}
We adopt the following metric ansatz 
\begin{equation}
	\label{eq:AdSAnsatz}
	ds^2 = \frac{l^2}{\cos^2 x}\left(- A(x) e^{-2\delta(x)} dt^2 + A(x)^{-1} dx^2 + \sin^2 x d\Omega^2_{d-1}\right),
\end{equation}
and the following scalar field ansatz
\begin{equation}
	\phi = \phi_0(x) e^{i\omega t}.
\end{equation}
The scalar field equation then becomes
\begin{equation}
	\phi_0'' + \left(\frac{A'}{A} - \delta + \frac{(d-1)}{\sin x \cos x}\right)\phi_0' + \frac{1}{A}\left(\frac{1}{A}e^{2\delta}\omega^2 - \frac{l^2}{\cos^2 x } U_0\right) \phi_0 = 0,\qquad U_0 = \frac{d V}{d |\phi|^2}.
\end{equation}
The stress-energy tensor becomes
\begin{subequations}
\begin{align}
	&\rho = -T^t_t = \frac{\cos^2 x}{l^2}\left(\frac{1}{A}e^{2\delta}\omega^2 \phi_0^2 + A(\phi_0')^2\right) + V,\\
	&p_r = T^x_x = \frac{\cos^2 x}{l^2}\left(\frac{1}{A}e^{2\delta}\omega^2 \phi_0^2 + A(\phi_0')^2\right) - V,\\
	&p_\theta = T^i_i = \frac{\cos^2 x}{l^2}\left(\frac{1}{A}e^{2\delta}\omega^2 \phi_0^2 - A(\phi_0')^2\right) - V,\qquad i \notin \{t, x\}.\\
\end{align}
\end{subequations}

\subsection{Metric decomposition}
Getting the metric part in arbitrary number of dimensions is harder. However, the following trick can be used. If the metric can be represented in the following form
\begin{equation}
g_{\mu\nu} =\begin{pmatrix}
	f_{ab}(x) & 0 \\
	0 & h_{ij}(y)
\end{pmatrix},
\end{equation}
when $f_{ab}$ is a $d_1$ dimensional metric which is a function of the first $d_1$ coordinates and $h_{ij}$ is $d_2$ dimensional metric which is a function of last $d_2$ coordinates, then Ricci tensor for this metric trivially becomes
\begin{equation}
	R_{\mu\nu} =\begin{pmatrix}
		R^f_{ab} & 0 \\
		0 & R^h_{ij}
	\end{pmatrix},
\end{equation}
where $R^f_{ab}$ is the Ricci tensor built purely on $f$ metric and $R^h$ is the Ricci tensor built purely on $h$ metric. By applying two successive conformal transformations we can get that the Ricci tensor for the metric 
\begin{equation}
	g_{\mu\nu} =\begin{pmatrix}
		f_{ab}(x) & 0 \\
		0 & e^{2\omega(x)} h_{ij}(y)
	\end{pmatrix},
\end{equation}
where $\omega(x)$ is a function of first $d_1$ coordinates only, can be expressed in a diagonal form
\begin{equation}
R_{\mu\nu} =\begin{pmatrix}
	R^1_{ab} & 0 \\
	0 & R^2_{ij}
\end{pmatrix},
\end{equation}
where
\begin{subequations}
\begin{align}
	&R^1_{ab} = R^f_{ab} - d_2(\triangledown_a \triangledown_b\omega + \partial_a\omega\partial_b\omega),\\
	&R^2_{ij} = R^h_{ij} - e^{2\omega} (\square\omega + d_2(\partial \omega)^2) h_{ij},
\end{align}
\end{subequations}
where all derivatives and all partial derivatives are taken with respect to first $d_1$ coordinates only.

We know that a $d-1$ dimensional sphere has the Ricci tensor
\begin{equation}
R^{sphere}_{ij} = (d - 2) h_{ij},
\end{equation}
therefore an arbitrary spherically symmetric spacetime with a metric
\begin{equation}
	ds^2 = f_{ab} dx^a dx^b + e^{2\omega} d\Omega^2_{d-1},
\end{equation}
has a Ricci tensor
\begin{subequations}
	\begin{align}
		&R^1_{ab} = R^f_{ab} - (d-1)(\triangledown_a \triangledown_b\omega + \partial_a\omega\partial_b\omega),\\
		&R^2_{ij} = ((d-2) - e^{2\omega} (\square\omega + (d-1)(\partial \omega)^2)) h_{ij},
	\end{align}
\end{subequations}
the Einstein tensor is
\begin{subequations}
\label{eq:GEinsteinDecomp}
\begin{align}
	&G^1_{ab} = G^f_{ab} - (d-1)(\triangledown_a\triangledown_b \omega + \partial_a\omega\partial_b \omega) +\\\nonumber
	&\qquad f_{ab} \left((d-1) \square\omega +\frac{1}{2}d(d-1)(\partial\omega)^2 - \frac{1}{2}(d-1)(d-2)e^{-2\omega}\right),\\
	&G^2_{ij} = h_{ij}\left[e^{2\omega}\left((d-2)\square\omega + \frac{1}{2}(d-1)(d-2)(\partial\omega)^2 - \frac{1}{2}R^f\right) - \frac{1}{2}(d-2)(d-3)\right],
\end{align}
\end{subequations}
where $R^f = f^{ab} R_{ab}^f$ is $f$ metric Ricci scalar.

\subsection{Gravitational equations}
Substituting the metric \eqref{eq:AdSAnsatz} into the equations \eqref{eq:GEinsteinDecomp} we obtain the following system of equations that fully describe the AdS Boson Star
\begin{subequations}
\label{eq:StarEOM}
\begin{align}
	&A'(x) = - \frac{d - 2 \cos^2 x}{\sin x \cos x}\left(A - 1\right) - \frac{2 \kappa \sin x \cos x}{d - 1} \left(\frac{e^{2\delta}}{A}\omega^2 \phi_0^2 + A(\phi_0')^2\right) - \frac{2 \kappa l^2 \tan x}{d - 1} V,\\
	&\delta'(x) = -\frac{2\kappa \sin x \cos x}{d - 1} \left(\frac{e^{2\delta}}{A^2}\omega^2 \phi_0^2 + (\phi_0')^2\right),\\
	\label{eq:ScalarEOM}
	&\phi_0'' + \left(\frac{A'}{A} - \delta + \frac{(d-1)}{\sin x \cos x}\right)\phi_0' + \frac{1}{A}\left(\frac{1}{A}e^{2\delta}\omega^2 - \frac{l^2}{\cos^2 x } U_0\right) \phi_0 = 0,\qquad U_0 = \frac{d V}{d |\phi|^2},
\end{align}
\end{subequations}

\subsection{Empty space solution}
If we assume that no star is present $\phi_0(x) = 0$, the equations \eqref{eq:StarEOM} admit the following empty space solution $\delta(x) = 0$ and
\begin{equation}
	A(x) = 1 - \frac{2 M}{l^{d-2}} \frac{\cos^2 x}{\tan^{d-2} x}.
\end{equation}
This solution corresponds to a $(d+1)$-dimensional Schwarzschild black hole with mass $M$ in AdS spacetime. By taking $r = l \tan x$ and $\tau = l t$ we can transform this solution to ordinary Schwarzschild coordinates
\begin{equation}
	ds^2 = - \left(1 - \frac{2M}{r^{d-2}} + \frac{r^2}{l^2}\right) d\tau^2 + \frac{dr^2}{1 - \frac{2M}{r^{d-2}} + \frac{r^2}{l^2}} + r^2 d \Omega^2_{d-1}.
\end{equation}

\section{Regular solutions for scalar field equation}
In the absence of gravity scalar field equation \eqref{eq:ScalarEOM} becomes
\begin{equation}
	\phi_0'' + \frac{(d-1)}{\sin x \cos x}\phi_0' + \left(\omega^2 - \frac{l^2}{\cos^2 x } U_0\right) \phi_0 = 0,\qquad U_0 = \frac{d V}{d |\phi|^2}.
\end{equation}
It is impossible to solve this equation in general, but it is possible to solve it in the linear case when $V = m^2 |\phi|^2$, the equation then becomes
\begin{equation}
	\phi_0'' + \frac{(d-1)}{\sin x \cos x}\phi_0' + \left(\omega^2 - \frac{l^2 m^2}{\cos^2 x }\right) \phi_0 = 0.
\end{equation}
The general solution of this equation is
\begin{equation}
\begin{split}
	\phi_0 = c_1 (\cos x)^{\alpha_+} F\left(\alpha_+ - \frac{\omega}{2}, \alpha_+ + \frac{\omega}{2}; 2\alpha_+ -\frac{d}{2}+1; \cos^2 x\right) + \\
	c_2 (\cos x)^{\alpha_-} F\left(\alpha_- - \frac{\omega}{2}, \alpha_- + \frac{\omega}{2}; 2\alpha_- -\frac{d}{2}+1; \cos^2 x\right),
\end{split}
\end{equation}
where $\alpha_\pm = \frac{1}{4}(d \pm \sqrt{d^2 + 4 l^2 m^2})$ and $F(a,b;c;x)$ is the hypergeometric function. The second branch is singular at the boundary $x = \pi/2$ so we are going to drop it. We can use the formula
\begin{equation}
\begin{split}
	F(a,b;c;x) =& \frac{\Gamma(c)\Gamma(c-a-b)}{\Gamma(c-a)\Gamma(c-b)}F(a,b;a+b+1-c; 1-x) +\\ &\frac{\Gamma(c)\Gamma(a+b-c)}{\Gamma(a)\Gamma(b)}(1-x)^{c-a-b}F(c-a,c-b;c+1-a-b,1-x),
\end{split}
\end{equation}
find conditions to make the solution regular at the origin. Since $c - a - b = 1 - d/2$, the regularity can only be achieved when $a = - j$ and $j$ is zero or a positive integer. In such case $1/\Gamma(a)$ becomes zero. Therefore if we take
\begin{equation}
	\omega = 2\alpha_+ + 2j,
\end{equation}
we obtain a solution regular everywhere
\begin{equation}
	\phi_0 = c_1 (\cos x)^{2\alpha_+} F\left(-j, 2\alpha_+ + j; 2\alpha_+ -\frac{d}{2}+1; \cos^2 x\right),
\end{equation}
the Hypergeometric function is a polynom in $\cos^2 x$. It is more convenient to rewrite this expression in terms of Jacobi polynomials
\begin{equation}
 	\phi_0 = C (\cos x)^{(d + \sqrt{d^2 + 4 l^2 m^2})/2} P^{d/2 - 1, \sqrt{d^2 + 4 l^2 m^2}/2}_j(\cos 2x),
\end{equation}
the corresponding frequency explicitly is
\begin{equation}
	\phi_0 = \frac{1}{2}(d + \sqrt{d^2 + 4 l^2 m^2}) + 2j.
\end{equation}
The fundamental mode corresponds to $j = 0$
\begin{equation}
	\phi_0 = C (\cos x)^{\frac{1}{2}(d + \sqrt{d^2 + 4 l^2 m^2})}
\end{equation}

\section{Asymptotic expansion}
\subsection{Boundary case $x = \pi/2$}
We introduce a new coordinate $\rho = \pi/2 - x$. If we expand the solution in terms or $\rho$ around $\rho = 0$ we get
\begin{subequations}
\begin{align}
	&A(\rho) = 1 + A^{(0)}\left(1 + \frac{d-3}{3}\rho^2\right) \rho^d + \frac{4\kappa n(n-1)}{(2n - d)(d-1)}(\phi^{(0)})^2 \rho^{2n} + O(\rho^{d+4}) + O(\rho^{2n+2}),\\
	&\delta = \frac{\kappa n\phi_0^2}{d-1} \rho^{2n} + O(\rho^{2n+2}),\\
	&\phi_0(\rho) = \phi^{(0)} \rho^{n} \left(1 + \frac{n(n + d - 2 - 3 n A^{(0)} \delta_{d,2} ) - 3\omega^2}{6(2(n+1) - d)} \rho^2\right) + O(\rho^{n+4}),
\end{align}
\end{subequations}
where we label $V'(0) = m^2$, $\delta_{d,2}$ is a Kronecker delta, and
\begin{equation}
	n = 2\alpha_+ = \frac{d}{2} + \frac{1}{2}\sqrt{d^2 + 4 l^2 m^2}
\end{equation}
We have absorbed $\delta(0)$ in the definition of $\omega$, $A^{(0)}$ and $\phi^{(0)}$ serve as free parameters. The system \eqref{eq:StarEOM} has a non-singular solution only when $V(0) = 0$.

These equations can be recast in the following form, allowing to obtain $\phi^{0}$ and $A^{(0)}$
\begin{subequations}
\begin{align}
	&A(\rho) - 1 = \left(1 + \frac{d-3}{3}\rho^2\right) (\rho^d A^{(0)}) + \frac{4\kappa n(n-1)}{(2n - d)(d-1)}(\rho^{n}\phi^{(0)})^2,\\
	&\phi_0(\rho) = \left(1 + \frac{n(n+d-2) - 3\omega^2}{6(2(n+1) - d)} \rho^2\right) (\rho^n\phi^{(0)}) - \frac{3 n^2 \delta_{d,2} }{6(2(n+1) - d)} (\rho^d A^{(0)})(\rho^{n}\phi^{(0)}).
\end{align}
\end{subequations}
We label $x = \rho^d A^{(0)}$ and $y = \rho^{n}\phi^{(0)}$ and 
\begin{equation}
\begin{split}
	&a_1 = 1 + \frac{d-3}{3}\rho^2,\quad a_2 = \frac{4\kappa n(n-1)}{(2n - d)(d-1)},\\
	&b_1 = \frac{n(n+d-2) - 3\omega^2}{6(2(n+1) - d)},\quad b_2 = \frac{3 n^2}{6(2(n+1) - d)}.
\end{split}
\end{equation}
The system then becomes
\begin{subequations}
\begin{align}
	&x = \frac{1}{a_1} (A(\rho) - 1 - a_2 y^2),\\
	&\delta_{d,2} \frac{a_2 b_2}{a_1} y^3 + \left(1 + b_1 \rho^2 - \delta_{d,2}\frac{b_2}{a_1}(A(\rho) - 1)\right)y - \phi_0(\rho)=0.
\end{align}
\end{subequations}
Finding $x$ and $y$ allows us to find solution around $x = \pi/2$ from the asymptotic as well as implement shooting by adjusting $\phi'_0(\rho)$
\begin{equation}
	\phi_0'(\rho) = \left(\frac{n}{\rho} + (n+2) b_1 \rho\right) y - \delta_{d,2} b_2 \frac{n + 2}{\rho} x y.
\end{equation}
